local function time(label, fn)
    local s = os.time()
    local n = 0
    repeat
        fn()
        n = n + 1
    until os.time() - s >= 1
    print(string.format("%-52s %9.4f ms per call (%d calls)", label, (os.time() - s) * 1000 / n, n))
end

    local r = Tables.deepCopy(start)
    item:getModData().tlgProc = r
    if serverMode then Vault.put("process", "item:7", r) end
end
local function passes(label, start)
    fresh(start)
    g.now = 600
    time("sampleSites, " .. label, function()
        g.now = g.now + 1
        Machines.sampleSites()
    end)
    fresh(start)
    g.now = 600
    time("observeFires, " .. label, function()
        g.now = g.now + 1 / 6
        Machines.observeFires()
    end)
end
