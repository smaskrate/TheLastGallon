module TLG
{
    /* Every icon here is the mod's own (Item_TLG_<Name>.png, rendered by
       tools/make-model-icons.ps1): the filter's from its model, the
       others' from simple shapes built for them
       (tools/blender/pz_icon_props.py). Display names come from
       Translate/<lang>/ItemName.json. */

    /* A01 sampling/testing kit: enables "kit" precision inspection. */
    item FuelSampleKit
    {
        DisplayCategory = VehicleMaintenance,
        ItemType = base:normal,
        Weight = 0.4,
        Icon = TLG_FuelSampleKit,
        Tags = base:glass,
    }

    /* One bottle treats 20 L at full strength (TLG.Domain.Treatment). */
    item FuelStabilizer
    {
        DisplayCategory = VehicleMaintenance,
        ItemType = base:normal,
        Weight = 0.3,
        Icon = TLG_FuelStabilizer,
    }

    /* Filter cartridge: sediment, free water and some biomass. On the floor
       it is the mod's own model (TLG_models.txt), and its icon is rendered
       from that model (tools/make-model-icons.ps1). */
    item FuelFilter
    {
        DisplayCategory = VehicleMaintenance,
        ItemType = base:normal,
        Weight = 0.2,
        Icon = TLG_FuelFilter,
        WorldStaticModel = TLG.FuelFilter,
    }

    /* Manual: read like the game's print media, it shows a page on
       start-up, fuel compatibility and troubleshooting of the ethanol loop
       (Integration/Manual.lua). It teaches no recipe. Found in garage, gas
       station, farm, generator crate, bookstore and library loot. */
    item ManualFuelSystems
    {
        DisplayCategory = SkillBook,
        ItemType = base:literature,
        Weight = 0.5,
        Icon = TLG_ManualFuelSystems,
        OnCreate = TLG.Integration.Manual.onCreate,
    }

    /* Saved from a finished ferment (>= 5 kg sugar): replaces bought yeast,
       so the ethanol chain has a renewable starter culture. */
    item YeastStarter
    {
        DisplayCategory = Food,
        ItemType = base:normal,
        Weight = 0.3,
        Icon = TLG_YeastStarter,
    }
}
