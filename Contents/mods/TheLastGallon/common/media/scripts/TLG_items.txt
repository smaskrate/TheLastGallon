module TLG
{
    /* Every icon and floor model here is the mod's own (TLG_models.txt).
       The icons (Item_TLG_<Name>.png) are rendered by
       tools/make-model-icons.ps1: the filter's from its supplied model,
       the others' from simple shapes built for them
       (tools/blender/pz_icon_props.py). The same shapes, as each lies on
       the floor, are the others' floor models (tools/make-prop-models.ps1).
       Display names come from Translate/<lang>/ItemName.json. */

    /* A01 sampling/testing kit: enables "kit" precision inspection. */
    item FuelSampleKit
    {
        DisplayCategory = VehicleMaintenance,
        ItemType = base:normal,
        Weight = 0.4,
        Icon = TLG_FuelSampleKit,
        WorldStaticModel = TLG.FuelSampleKit,
        Tags = base:glass,
    }

    /* One bottle treats 20 L at full strength (TLG.Domain.Treatment). */
    item FuelStabilizer
    {
        DisplayCategory = VehicleMaintenance,
        ItemType = base:normal,
        Weight = 0.3,
        Icon = TLG_FuelStabilizer,
        WorldStaticModel = TLG.FuelStabilizer,
    }

    /* Filter cartridge: sediment, free water and some biomass. On the floor
       it is the mod's own model (TLG_models.txt), and its icon is rendered
       from that model (tools/make-model-icons.ps1). */
    item FuelFilter
    {
        DisplayCategory = VehicleMaintenance,
        ItemType = base:normal,
        Weight = 0.2,
        Icon = TLG_FuelFilter,
        WorldStaticModel = TLG.FuelFilter,
    }

    /* Manual: read like the game's print media, it shows a page on
       start-up, fuel compatibility and troubleshooting of the ethanol loop
       (Integration/Manual.lua). It teaches no recipe. Found in garage, gas
       station, farm, generator crate, bookstore and library loot. */
    item ManualFuelSystems
    {
        DisplayCategory = SkillBook,
        ItemType = base:literature,
        Weight = 0.5,
        Icon = TLG_ManualFuelSystems,
        WorldStaticModel = TLG.ManualFuelSystems,
        OnCreate = TLG.Integration.Manual.onCreate,
    }

    /* The biodiesel handbook (0.4.0): reading it teaches the recipes of the
       five biodiesel machines (TLG_machines.txt), as a vanilla recipe
       magazine does. Found where the field manual is (Integration/Manual.lua).
       On the ground it uses the field manual's model. */
    item BiodieselHandbook
    {
        DisplayCategory = RecipeResource,
        ItemType = base:literature,
        Weight = 0.5,
        Icon = TLG_BiodieselHandbook,
        BoredomChange = -20,
        StressChange = -15,
        LearnedRecipes = TLG_MakeWoodRetort;TLG_MakeAshBarrel;TLG_MakeLyeLeacher;TLG_MakeBiodieselReactor;TLG_MakeBiodieselWashTank,
        StaticModel = Magazine,
        WorldStaticModel = TLG.ManualFuelSystems,
        Tags = base:magazine,
        OnCreate = ItemCodeOnCreate.onCreateRecipeMagazine,
    }

    /* The wood gas handbook (D-040): reading it teaches the gasifier's
       recipe (TLG_machines.txt), as the biodiesel handbook teaches its
       machines'. Found where the field manual is (Integration/Manual.lua).
       On the ground it uses the field manual's model. */
    item WoodGasHandbook
    {
        DisplayCategory = RecipeResource,
        ItemType = base:literature,
        Weight = 0.5,
        Icon = TLG_WoodGasHandbook,
        BoredomChange = -20,
        StressChange = -15,
        LearnedRecipes = TLG_MakeGasifier,
        StaticModel = Magazine,
        WorldStaticModel = TLG.ManualFuelSystems,
        Tags = base:magazine,
        OnCreate = ItemCodeOnCreate.onCreateRecipeMagazine,
    }

    /* Saved from a finished ferment (>= 5 kg sugar): replaces bought yeast,
       so the ethanol chain has a renewable starter culture. */
    item YeastStarter
    {
        DisplayCategory = Food,
        ItemType = base:normal,
        Weight = 0.3,
        Icon = TLG_YeastStarter,
        WorldStaticModel = TLG.YeastStarter,
    }
}
