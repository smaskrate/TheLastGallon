module TLG
{
    /* The mod's own models (docs/asset-audit.csv, art-src/).

       Fermenter: the fermentation vessel on the floor. Mesh
       media/models_X/TLG/Fermenter.fbx and texture
       media/textures/TLG/Fermenter.png, supplied 2026-11-02 (provenance in
       art-src/fermenter/PROVENANCE.md). The FBX is Y up with its base at
       the origin and 1.0 tall. The game draws an FBX at its vertices times
       the node's own scaling times scale times 1.5 (vanilla WaterTank and
       CookingPot measured against a character, run 66; the 1.5 is what
       undoCoreScale takes back, docs/engine-capabilities.md G13), so 0.5
       stands it about 0.75 m tall: a 160 L vessel. */
    model Fermenter
    {
        mesh = TLG/Fermenter,
        texture = TLG/Fermenter,
        scale = 0.5,
    }

    /* FermenterHeld: the same barrel carried in both arms. The carry clip
       (Bob_TLG_CarryVessel, AnimSets/player/maskingright/tlgcarryvessel.xml)
       puts Bip01_Prop1 in front of the chest, turned so this attachment
       stands the barrel upright with its tap forward: the hurricane
       lantern's Bip01_Prop1 turn (models_items.txt
       Lantern_Hurricane_HandleUp). An attachment offset moves the model
       1.5 m per unit whatever its scale, so -0.25 centres the 0.75 m barrel
       on the bone (RT-SP runs 86-95). */
    model FermenterHeld
    {
        mesh = TLG/Fermenter,
        texture = TLG/Fermenter,
        scale = 0.5,
        attachment Bip01_Prop1
        {
            offset = 0.0 0.0 -0.25,
            rotate = -94.8078 -85.9541 174.4184,
        }
    }

    /* The pot still, the reflux still, the drying column and the fuel
       filter, supplied 2026-11-03 (provenance in art-src/potstill,
       refluxstill, dryingcolumn and fuelfilter): the same kind of FBX as
       the fermenter, Y up, base at the origin, each 1.0 along its largest
       side. Each scale sets a true size from the capacity the machine
       models (docs/asset-requirements.md). PotStill: a 60 L pot still,
       0.42 of the model across its pot, so 0.7 gives a 0.44 m pot about
       0.36 m deep (55 L); the swan neck and hose take it to 1.05 m wide
       and 0.68 m tall. */
    model PotStill
    {
        mesh = TLG/PotStill,
        texture = TLG/PotStill,
        scale = 0.7,
    }

    /* RefluxStill: a 120 L boiler, 0.41 across in the model, so 0.9 gives
       a 0.55 m pot 0.46 m deep (110 L) under its column, 1.35 m tall in
       all. */
    model RefluxStill
    {
        mesh = TLG/RefluxStill,
        texture = TLG/RefluxStill,
        scale = 0.9,
    }

    /* DryingColumn: the column stands in its receiver jar, 0.425 across in
       the model, so 0.55 gives a 0.35 m jar 0.40 m tall (40 L) and 0.82 m
       in all. */
    model DryingColumn
    {
        mesh = TLG/DryingColumn,
        texture = TLG/DryingColumn,
        scale = 0.55,
    }

    /* FuelFilter: a filter canister 0.15 m tall and 0.14 m across. None of
       these four has a hand model: the game draws a held non-weapon only
       during an action, and at true size a still held then covered the
       character from feet to head (RT-SP runs 120-121); with no hand model
       nothing is drawn there, as for a carried generator. */
    model FuelFilter
    {
        mesh = TLG/FuelFilter,
        texture = TLG/FuelFilter,
        scale = 0.1,
    }

    /* The sampling kit, the yeast starter, the fuel stabilizer and the
       manual on the floor: the mod's own shapes, the ones their icons are
       rendered from, built as each lies on the floor by
       tools/make-prop-models.ps1 (tools/blender/pz_prop_models.py). Each
       is the same kind of FBX as the supplied models, 1.0 along its
       largest side with its base at the origin, and has a texture of flat
       colours. FuelSampleKit: three sample jars with a hose coiled flat in
       front; 0.225 gives jars 0.12 m tall (0.13 m with their lids) and the
       kit 0.34 m across and deep. */
    model FuelSampleKit
    {
        mesh = TLG/FuelSampleKit,
        texture = TLG/FuelSampleKit,
        scale = 0.225,
    }

    /* YeastStarter: a jar of culture under a cloth tied with string; 0.108
       gives a jar 0.13 m across, 0.16 m tall with its cloth. */
    model YeastStarter
    {
        mesh = TLG/YeastStarter,
        texture = TLG/YeastStarter,
        scale = 0.108,
    }

    /* FuelStabilizer: a corked bottle of tar; 0.115 stands it 0.17 m tall
       and 0.11 m across. */
    model FuelStabilizer
    {
        mesh = TLG/FuelStabilizer,
        texture = TLG/FuelStabilizer,
        scale = 0.115,
    }

    /* ManualFuelSystems: the field manual on its back; 0.16 makes it
       0.24 m by 0.18 m and 2.4 cm thick. */
    model ManualFuelSystems
    {
        mesh = TLG/ManualFuelSystems,
        texture = TLG/ManualFuelSystems,
        scale = 0.16,
    }
}
